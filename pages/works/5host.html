コードでのUIアニメーション制御

部屋名UIとセリフUIのフェードイン/アウトアニメーションを実装しました。
エンジンの機能に頼らず、プログラムで座標や透明度を計算して制御しています。

RoomNameBox.cs, RoomNameText.cs
※どちらもほぼ同じ処理 

          //関数:アニメーション再生.
          void AnimationPlay()
          {
              //※見やすくする用.
              Vector3 pos = transform.localPosition;

              //移動.
              transform.localPosition += new Vector3(roomname.box.MoveImage_x, 0, 0);

              //x軸が一定ラインまで達したら.
              if (pos.x >= roomname.SlowDownLineX)
              {
                  //減速.
                  roomname.box.MoveImage_x -= roomname.SlowDown;

                  //マイナスに逆ブレしたら0にする.
                  if (roomname.box.MoveImage_x < 0) {
                      roomname.box.MoveImage_x = 0;
                  }
              }

              //停止したら.
              if (roomname.box.MoveImage_x == 0)
              {
                  roomname.box.Move_Flg = false; //アニメーション終了.
              }
          }
RoomNameManager.cs 

          //関数:フェードアウト実行.
          void RoomName_Fadeout()
          {
              //表示アニメーション.
              if (color_a >= 0.0f)
              {
                  /*
                    ゆっくり消滅。
                  */

                  hie_RoomNameBox.GetComponent<Image>().color = new Color(255, 255, 255, color_a);
                  hie_RoomNameText.GetComponent<Text>().color = new Color(0, 0, 0, color_a);

                  color_a -= FadeoutSpeed; //透明度を上げる.
              }
              //アニメーション終了.
              else
              {
                  //非表示.
                  hie_RoomNameBox.SetActive(false);
                  hie_RoomNameText.SetActive(false);

                  Fadeout_Flg = false;
              }
          }

          //アニメーション実行.
          public void AnimationExe()
          {
              CancelInvoke();

              //▼アニメーション実行▼

              /*
                カウントは同時に始まる。
              */

              //アニメーション実行.
              Invoke("Exe_RoomNameBox",  0.0f);
              Invoke("Exe_RoomNameText", 0.2f);

              //フェードアウト(消滅)
              Invoke("Fadeout_RoomName", 3.5f);
          }

          void Exe_RoomNameBox()
          {
              box.Move_Flg = true; //RoomNameBoxのAnimation実行.
          }

          void Exe_RoomNameText()
          {
              text.Move_Flg = true; //RoomNameTextのAnimation実行.
          }

          void Fadeout_RoomName()
          {
              Fadeout_Flg = true;
              color_a = 1f; //最初は透過なし.
          }

SerifBox.cs 

          void Serif_Fadein()
          {
              //初回のみ.
              if (color_a == 0)
              {
                  hie_SerifBox.SetActive(true); //表示.
              }

              //表示アニメーション.
              if (color_a < 1.0f)
              {
                  Serif_y += 1; //bpxを上に動かす.

                  /*
                    ゆっくり表示。
                    (いくつかやり方はあるが、今回はグループを使ってみる)
                  */
                  //セリフboxやキャラUIを、グループにする.
                  CanvasGroup SerifBoxGroup = hie_SerifBox.GetComponent<CanvasGroup>();
                  //↑のグループ全ての透明度を変更.
                  SerifBoxGroup.alpha = color_a;

                  //セリフboxを動かす.
                  hie_SerifBox.transform.localPosition = new Vector2(Serif_x, Serif_y);

                  color_a += 0.08f; //透明度を下げる.
              }
              //アニメーション終了.
              else
              {
                  color_a = 1.0f; //固定.
                  Fadein_Flg = false;
              }
          }

          void Serif_Fadeout()
          {
              //表示アニメーション.
              if (color_a > 0.0f)
              {
                  Serif_y -= 1; //boxを下に動かす.

                  /*
                    ゆっくり消滅。
                  */
                  //セリフboxやキャラUIを、グループにする.
                  CanvasGroup SerifBoxGroup = hie_SerifBox.GetComponent();
                  //↑のグループ全ての透明度を変更.
                  SerifBoxGroup.alpha = color_a;

                  //セリフboxを動かす.
                  hie_SerifBox.transform.localPosition = new Vector2(Serif_x, Serif_y);
                  /*
                    spriteはtransform.position(ワールド座標)で使えたが
                    canvasでやる場合は、transform.localPosition(ローカル座標)を
                  */

                  color_a -= 0.08f; //透明度を上げる.
              }
              //アニメーション終了.
              else
              {
                  hie_SerifBox.SetActive(false); //非表示.

                  Fadeout_Flg = false;
                  color_a = 0.0f; //リセット.
              }
          }
